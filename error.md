def read_txt_landmarks(label_path):
    pts = {}
    with open(label_path, "r") as f:
        for ln_no, line in enumerate(f, 1):
            line = line.strip()
            if not line:
                continue
            parts = [p.strip() for p in line.replace("\t", ",").split(",") if p.strip() != ""]
            if len(parts) != 3:
                print(f"[WARN] Bad line in {label_path}:{ln_no}: '{line}'")
                return None
            tag, sx, sy = parts
            try:
                x, y = float(sx), float(sy)
            except Exception:
                print(f"[WARN] Non-numeric coord in {label_path}:{ln_no}: '{line}'")
                return None
            pts[tag] = (x, y)
    missing = [n for n in LANDMARK_NAMES if n not in pts]
    extra = [n for n in pts.keys() if n not in LANDMARK_NAMES]
    if missing or extra:
        print(f"[WARN] Label issue in {label_path} | missing={missing} | extra={extra}")
        return None
    coords = []
    for name in LANDMARK_NAMES:
        coords.extend(pts[name])
    return coords

def build_dataset_lists(image_folder, label_folder):
    image_paths = []
    keypoints = []
    for img_name in sorted(os.listdir(image_folder)):
        if not img_name.lower().endswith(".png"):
            continue
        base = os.path.splitext(img_name)[0]
        label_path = os.path.join(label_folder, f"{base}.txt")
        if not os.path.exists(label_path):
            print(f"[WARN] Missing label for {img_name}")
            continue
        kp = read_txt_landmarks(label_path)
        if kp is None:
            continue
        if len(kp) != NUM_LANDMARKS * 2:
            print(f"[WARN] Wrong number of coords for {img_name}")
            continue
        image_paths.append(os.path.join(image_folder, img_name))
        keypoints.append(kp)
    return image_paths, keypoints

class XrayLandmarkDataset(Dataset):
    def __init__(self, image_paths, keypoints, transform=None):
        self.image_paths = image_paths
        self.keypoints = keypoints
        self.transform = transform

    def __len__(self):
        return len(self.image_paths)

    def __getitem__(self, idx):
        image = cv2.imread(self.image_paths[idx], cv2.IMREAD_GRAYSCALE)

        # convert to 3 channel
        image = np.stack([image, image, image], axis=-1)
        kp_flat = self.keypoints[idx]
        keypoints = [(kp_flat[i], kp_flat[i + 1]) for i in range(0, len(kp_flat), 2)]
        if self.transform:
            augmented = self.transform(image=image, keypoints=keypoints)
            image = augmented["image"]
            keypoints = augmented["keypoints"]
        _, h, w = image.shape
        targets = []
        mask = []
        for (x, y) in keypoints:
            valid = (x is not None) and (y is not None) and (0 <= x < w) and (0 <= y < h)
            if valid:
                tx = float(np.clip(x / w, 0.0, 1.0))
                ty = float(np.clip(y / h, 0.0, 1.0))
                mask.append(1.0)
            else:
                tx, ty = 0.0, 0.0
                mask.append(0.0)
            targets.extend([tx, ty])
        targets = torch.tensor(targets, dtype=torch.float32)
        mask = torch.tensor(mask, dtype=torch.float32)
        return image, targets, mask

class CoordRegressor(nn.Module):
    def __init__(self, base):
        super().__init__()
        self.base = base

    def forward(self, x):
        x = self.base(x)
        x = torch.sigmoid(x)
        return x

# --- CORRECTED WINGLOSS PARAMETERS FOR [0, 1] NORMALIZED TARGETS ---
class WingLoss(nn.Module):
    def __init__(self, w, epsilon):
        super().__init__()
        self.w = w
        self.epsilon = epsilon
        self.C = self.w - self.w * math.log(1.0 + self.w / self.epsilon)

    def forward(self, prediction, target):
        y = torch.abs(prediction - target)
        loss = torch.where(y < self.w, self.w * torch.log(1.0 + y / self.epsilon), y - self.C)
        return loss

class MultifacetedLoss(nn.Module):
    def __init__(self, w=0.1, epsilon=0.02):  # 1.5 ,08  < .85 , .1 current setup..
        super().__init__()
        self.wing_loss = WingLoss(w, epsilon)

    def forward(self, prediction, target, mask=None):
        B, dim = prediction.shape
        L = dim // 2
        loss = self.wing_loss(prediction, target).view(B, L, 2).mean(dim=2)
        if mask is not None:
            valid_sum = mask.sum().clamp_min(1.0)
            loss = (loss * mask).sum() / valid_sum
        else:
            loss = loss.mean()
        return loss
